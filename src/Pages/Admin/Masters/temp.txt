import React, { useState, useMemo } from 'react';
import { Link } from 'react-router-dom';
import SidebarIcon from './SidebarIcon'; // Import the new SidebarIcon component

const Sidebar = ({ role, RoutesLink }) => {
    const [isActive, setIsActive] = useState('');
    const [sites, setSites] = useState(false);

    const toggleIsActive = (name) => setIsActive(name);
    const toggleSites = () => setSites(!sites);

    const menuItems = useMemo(() => [
        {
            name: 'dashboard',
            label: 'Dashboard',
            route: RoutesLink?.dashboard_route,
            iconPath: "m460.414 216.518-85.6-73.1c...",
        },
        {
            name: 'sites',
            label: 'Sites',
            route: RoutesLink?.sites_route,
            iconPath: "M18.179 39.571A1.002 1.002...",
            subMenu: [
                {
                    name: 'vendor',
                    label: 'Vendors',
                    route: RoutesLink?.vendors_route,
                    iconPath: "m498.52 219.49-52.86-20.46c...",
                },
            ],
        },
    ], [RoutesLink]);

    return (
        role === 'admin' && (
            <div className="sidebar-inner">
                <nav className="aside-nav">
                    <ul className="sidebar-vertical">
                        <li><span className="menu-title">Menu</span></li>

                        {menuItems.map((item) => (
                            <li key={item.name}>
                                <Link 
                                    to={item.route}
                                    onClick={() => toggleIsActive(item.name)}
                                    className={isActive === item.name ? "drop-link active" : "drop-link"} 
                                    role="button"
                                >
                                    <SidebarIcon path={item.iconPath} />
                                    <span>{item.label}</span>
                                </Link>

                                {item.subMenu && (
                                    <>
                                        <button className="drop-link btn-menu" onClick={toggleSites} role="button">
                                            <span className="menu-title border-0 p-0 m-0">{item.label}</span>
                                            <svg className={sites ? "chevron ms-auto dropdown rotate" : "chevron ms-auto dropdown"} xmlns="http://www.w3.org/2000/svg" width="20" height="20" viewBox="0 0 20 20">
                                                <path d="M6.4 8.2L10 11.8 13.6 8.2" stroke="#818898" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
                                            </svg>
                                        </button>

                                        <ul className="sub-menu" style={{ display: sites ? 'block' : 'none' }}>
                                            {item.subMenu.map((subItem) => (
                                                <li key={subItem.name}>
                                                    <Link
                                                        to={subItem.route}
                                                        onClick={() => toggleIsActive(subItem.name)}
                                                        className={isActive === subItem.name ? "drop-link active" : "drop-link"}
                                                        role="button"
                                                    >
                                                        <SidebarIcon path={subItem.iconPath} />
                                                        <span>{subItem.label}</span>
                                                    </Link>
                                                </li>
                                            ))}
                                        </ul>
                                    </>
                                )}
                            </li>
                        ))}
                    </ul>
                </nav>
            </div>
        )
    );
};

export default Sidebar;
